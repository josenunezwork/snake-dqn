def load(path):
    return json.loads(path.read_text())


def bind(path):
    assert path.is_absolute() and path.is_file() and not path.is_symlink(), path
    with path.open("rb") as handle:
        digest = hashlib.file_digest(handle, "sha256").hexdigest()
    return {"path": str(path), "sha256": digest, "size": path.stat().st_size}


def git(*args):
    return subprocess.check_output(["git", *args], cwd=BROKER, text=True).strip()


def run(argv):
    result = subprocess.run([str(a) for a in argv], cwd=BROKER, capture_output=True, text=True, timeout=90)
    return {"argv": [str(a) for a in argv], "return_code": result.returncode, "stdout": result.stdout, "stderr": result.stderr}
