def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def load(path):
    return json.loads(path.read_text())


def sha(path):
    with path.open("rb") as handle:
        return hashlib.file_digest(handle, "sha256").hexdigest()


def bind(path):
    assert path.is_absolute() and path.is_file() and not path.is_symlink(), path
    return {"path": str(path), "sha256": sha(path), "size": path.stat().st_size}


def run(argv, timeout=60):
    started = now()
    result = subprocess.run([str(a) for a in argv], cwd=BROKER, capture_output=True,
                            text=True, timeout=timeout)
    row = {"argv": [str(a) for a in argv], "started_utc": started,
           "finished_utc": now(), "return_code": result.returncode,
           "stdout": result.stdout, "stderr": result.stderr}
    assert result.returncode == 0, row
    return row


def git(*args):
    return run(["git", *args])["stdout"].strip()


def verify_reference(reference):
    observed = bind(Path(reference["path"]))
    assert observed["sha256"] == reference["sha256"], reference
    assert observed["size"] == reference.get("size", reference.get("size_bytes")), reference
    return observed


def account_preflight():
    accounts = ROOT / "accounts"
    accounts.mkdir()
    seed = accounts / "seed.json"
    export_json(seed, {"budget": {"other_assignments": {}}})
    with dispatch.allocation_lock(OPS / "alloc.lock", 60) as deadline:
        before = run(["/Users/josenunez/.local/bin/colab", "--auth", "oauth2", "sessions"])
        refreshed = dispatch.refresh_assignments(seed, accounts / "broker-session-review.json", deadline)
        rates = load(refreshed)["budget"]["other_assignments"]
        usage = run(["/Users/josenunez/.local/bin/colab", "--auth", "oauth2", "usage"])
        after = run(["/Users/josenunez/.local/bin/colab", "--auth", "oauth2", "sessions"])
        assert before["stdout"] == after["stdout"], "sessions changed during preflight"
        numbers = {}
        for key, label in (("balance", "Current balance"), ("rate", "Usage rate"),
                           ("assignments", "Active assignments")):
            match = re.search(rf"{label}:\s*([0-9]+(?:\.[0-9]+)?)", usage["stdout"])
            assert match, usage
            numbers[key] = float(match[1])
        account_dir = Path.home() / ".config/colab-cli/snake"
        pending = [bind(p) for p in sorted(account_dir.glob("pending-*.json"))]
        active = [bind(p) for p in sorted(account_dir.glob("active-*.json"))]
        assert not pending, "unresolved pending broker assignment"
        assert numbers["assignments"] == len(rates) < 3
        assert abs(numbers["rate"] - sum(rates.values())) < 0.05
        assert len(active) == len(rates), "unresolved active broker record"
        committed = sum(load(Path(row["path"]))["unit_cap"] for row in active)
        assert numbers["balance"] > 100 + committed
        value = {"schema": "bl-f5-canary02-account-preflight/v1", "status": "PASS",
                 "observed_utc": now(), "normal_allocation_lock": str(OPS / "alloc.lock"),
                 "calls": [before, usage, after], "other_assignments": rates,
                 "broker_session_review": bind(accounts / "broker-session-review.json"),
                 "usage": numbers, "active_records": active, "pending_records": pending,
                 "committed_units": committed, "account_assignment_cap": 3,
                 "allocation_calls": 0, "dispatch_refresh_required": True}
    export_json(accounts / "preflight.json", value)
    return rates
