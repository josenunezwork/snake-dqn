def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha(path):
    with path.open("rb") as handle:
        return hashlib.file_digest(handle, "sha256").hexdigest()


def bind(path):
    assert path.is_absolute() and path.is_file() and not path.is_symlink(), path
    return {"path": str(path), "sha256": sha(path), "size": path.stat().st_size}


def load(path):
    return json.loads(path.read_text())


def run(argv, *, timeout=60, env=None):
    started = now()
    result = subprocess.run([str(a) for a in argv], cwd=BROKER, env=env,
                            capture_output=True, text=True, timeout=timeout)
    row = {"argv": [str(a) for a in argv], "started_utc": started,
           "finished_utc": now(), "return_code": result.returncode,
           "stdout": result.stdout, "stderr": result.stderr}
    assert result.returncode == 0, row
    return row


def git(*args):
    return run(["git", *args])["stdout"].strip()


def verify_reference(reference):
    actual = bind(Path(reference["path"]))
    assert actual["sha256"] == reference["sha256"], reference
    expected_size = reference.get("size", reference.get("size_bytes"))
    assert expected_size is None or actual["size"] == expected_size, reference
    return actual
