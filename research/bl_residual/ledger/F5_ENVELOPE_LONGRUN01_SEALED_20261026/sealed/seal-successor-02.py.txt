def load(p):
    return json.loads(p.read_bytes())

def bind(p):
    assert p.is_absolute() and p.is_file() and not p.is_symlink(), p
    with p.open('rb') as f:
        sha = hashlib.file_digest(f, 'sha256').hexdigest()
    return dict(path=str(p), sha256=sha, size=p.stat().st_size)

def git(*args):
    return subprocess.check_output(['git', *args], cwd=BROKER, text=True, timeout=30).strip()

def verify_seal(root):
    p = root / 'SHA256SUMS.json'
    verify_export(p)
    seal = load(p)
    expected = set(seal['files']) | set(seal['excluded_self_components'])
    actual = {p.relative_to(root).as_posix() for p in root.rglob('*') if p.is_file()}
    assert actual == expected, (actual - expected, expected - actual)
    for name, row in seal['files'].items():
        observed = bind(root / name)
        assert (observed['sha256'], observed['size']) == (row['sha256'], row['size']), name
    return seal

def refs(obj, found):
    if isinstance(obj, dict):
        if isinstance(obj.get('path'), str) and 'sha256' in obj:
            p = Path(obj['path'])
            if p.is_absolute():
                row = bind(p)
                assert row['sha256'] == obj['sha256'], p
                if 'size' in obj:
                    assert row['size'] == obj['size'], p
                if 'size_bytes' in obj:
                    assert row['size'] == obj['size_bytes'], p
                found[str(p)] = row
        for v in obj.values():
            refs(v, found)
    elif isinstance(obj, list):
        for v in obj:
            refs(v, found)

def differences(a, b, pointer=''):
    if isinstance(a, dict) and isinstance(b, dict):
        result=[]
        for k in sorted(set(a) | set(b)):
            p = pointer + '/' + k.replace('~', '~0').replace('/', '~1')
            if k not in a or k not in b:
                result.append(dict(pointer=p, before=a.get(k), after=b.get(k)))
            else:
                result.extend(differences(a[k], b[k], p))
        return result
    if isinstance(a, list) and isinstance(b, list) and len(a)==len(b):
        return [d for i,(x,y) in enumerate(zip(a,b)) for d in differences(x,y,pointer+'/'+str(i))]
    return [] if a==b else [dict(pointer=pointer, before=a, after=b)]

def publish():
    seal=verify_seal(ROOT)
    target=REPO/'research/bl_residual/ledger/F5_ENVELOPE_LONGRUN01_SEALED_20261026'
    assert not target.exists()
    target.mkdir()
    records={}
    for p in sorted(ROOT.rglob('*')):
        if not p.is_file():continue
        row=bind(p); name=p.relative_to(ROOT).as_posix()
        record=dict(external_path=str(p),sha256=row['sha256'],size=row['size'])
        if row['size']>2_000_000:
            record['publication']='EXTERNAL_HASH_BOUND_ARTIFACT'
        else:
            relative='sealed/'+name+('.txt' if name.endswith('.py') else '')
            destination=target/relative
            destination.parent.mkdir(parents=True,exist_ok=True)
            with destination.open('xb') as f:f.write(p.read_bytes())
            assert bind(destination)['sha256']==row['sha256']
            record['published_relative_path']=relative
        records[name]=record
    export_json(target/'ARTIFACT_PACKET.json',dict(schema='bl-f5-longrun01-successor-publication/v1',package_root=str(ROOT),exhaustive_seal=bind(ROOT/'SHA256SUMS.json'),runtime_source_commit=PIN,files=records,script_snapshots='Byte-identical .py.txt; no runtime changes',allocations=0,dispatches=0,provider_calls=0))
    print(json.dumps(dict(publication=bind(target/'ARTIFACT_PACKET.json'),files=len(records)),indent=2))
