def load(path):
    return json.loads(path.read_text())


def bind(path):
    assert path.is_absolute() and path.is_file() and not path.is_symlink(), path
    with path.open("rb") as handle:
        digest = hashlib.file_digest(handle, "sha256").hexdigest()
    return {"path": str(path), "sha256": digest, "size": path.stat().st_size}


def git(*args):
    return subprocess.check_output(["git", *args], cwd=BROKER, text=True).strip()

def seal():
    assert not (ROOT / "SHA256SUMS.json").exists()
    prep = load(ROOT / "PREPARATION.json")
    m = load(ROOT / "manifest.json")
    verified = load(ROOT / "VERIFICATION.json")
    local = load(ROOT / "LOCAL_REVIEW.json")
    assert local["status"] == "NO_BLOCKING_FINDINGS"
    assert verified["return_code"] == 0 and verified["effective_niceness"] == 15
    assert all(v == "1" for v in verified["numeric_threads"].values())
    assert "255 passed, 2 skipped" in (ROOT / "verified-tests.log").read_text()
    assert not git("status", "--porcelain", "--untracked-files=all")
    assert not git("branch", "--show-current") and git("rev-parse", "HEAD") == m["commit"]
    references = load(ROOT / "bindings.json")
    for reference in references.values():
        assert bind(Path(reference["path"])) == reference
    registry = load(Path(references["qualified_registry"]["path"]))
    assert qualification.check(BROKER, Path(references["qualified_registry"]["path"])) == prep["cli_tree"]
    admission.require_final_only(load(Path(registry["acceptance_result"])), BROKER)
    for reference in (prep["manifest"], prep["source_archive"], prep["bindings"], prep["account_preflight"], prep["dry_run"], verified["log"]):
        assert bind(Path(reference["path"])) == reference
    for item in m["inputs"]:
        if "path" in item:
            observed = bind(Path(item["path"]))
            assert observed["sha256"] == item["sha256"] and observed["size"] == item["size_bytes"]
    control = load(ROOT / "inputs/control.json")
    assert control["source_hashes"] == f5_package.source_inventory(BROKER)
    f5_job._source_admission(control)
    binding = f5_timing_resume.identity(namespace="bl-residual-f5-timing-recovery-20261026",
        allocation=control["identity"]["allocation_sha256"], source_sha256=control["identity"]["source_sha256"],
        environment_sha256=control["identity"]["environment_sha256"], input_sha256=control["timing_input_sha256"])
    assert not f5_timing_resume.load_ledger(ROOT / "inputs/verified-ledger.json", binding)
    dispatch.require_resume_policy(m)
    dispatch.require_scientific_or_f5_plan_admission(m)
    supervisor.validate(m, BROKER)
    for path in prep["fresh_paths"]:
        assert not Path(path).exists(), path
    for path in ROOT.rglob("*.complete"):
        verify_export(Path(str(path).removesuffix(".complete")))
    export_json(ROOT / "PACKAGE.json", {
        "schema": "bl-f5-step8-family-canary-package/v1", "status": "SEALED_PENDING_ASTRA_OPERATIONAL_REVIEW",
        "created_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "source_commit": m["commit"], "whole_tree": prep["whole_tree"], "cli_tree": prep["cli_tree"],
        "transport_qualified_commit": registry["qualified_commit"], "registry_unchanged": True,
        "manifest": bind(ROOT / "manifest.json"), "preparation": bind(ROOT / "PREPARATION.json"),
        "bindings": bind(ROOT / "bindings.json"), "verification": bind(ROOT / "VERIFICATION.json"),
        "accounting": bind(ROOT / "ACCOUNTING_APPENDIX.json"), "dry_run": bind(ROOT / "dry-run.json"),
        "local_requirements_review": bind(ROOT / "LOCAL_REVIEW.json"),
        "source_archive": bind(ROOT / "source.tar"), "source_archive_members_verified": prep["source_archive_members_verified"],
        "inventory_sha256": prep["inventory_sha256"], "runtime_input_destinations": len(m["inputs"]),
        "dispatch_instructions": bind(ROOT / "DISPATCH.txt"), "dispatch_argv": prep["dispatch_argv"],
        "allocations": 0, "dispatches": 0, "canary_receipt_exists": False,
        "canary01_authorization_consumed": True, "successor_dispatch_authorized": False,
        "canary_producer_records": "CANARY_ONLY; fixture672 exported, scanner verifies ledger, revisit skips672, then exports673; live proof pending",
        "required_next_review": "Separate Astra operational review of this concrete successor before operator dispatch. Local reviewer does not grant Astra approval.",
    })
    members = {}
    for path in sorted(ROOT.rglob("*")):
        assert not path.is_symlink(), path
        if path.is_file():
            observed = bind(path)
            members[path.relative_to(ROOT).as_posix()] = {"size": observed["size"], "sha256": observed["sha256"]}
    self_parts = ["SHA256SUMS.json", "SHA256SUMS.json.sha256", "SHA256SUMS.json.complete"]
    result = export_json(ROOT / "SHA256SUMS.json", {
        "schema": "bl-f5-exhaustive-static-package-seal/v1", "scope": str(ROOT),
        "files": members, "excluded_self_components": self_parts,
        "fresh_output_state_outside_static_membership": prep["fresh_paths"],
    })
    verify_export(ROOT / "SHA256SUMS.json")
    assert {p.relative_to(ROOT).as_posix() for p in ROOT.rglob("*") if p.is_file()} == set(members) | set(self_parts)
    for name, row in members.items():
        observed = bind(ROOT / name)
        assert observed["sha256"] == row["sha256"] and observed["size"] == row["size"]
    print(json.dumps({"seal": result, "members": len(members), "manifest": bind(ROOT / "manifest.json"),
                      "package": bind(ROOT / "PACKAGE.json")}, indent=2))


if __name__ == "__main__":

        accounting()
    else:
        assert not sys.argv[1:]
        seal()
