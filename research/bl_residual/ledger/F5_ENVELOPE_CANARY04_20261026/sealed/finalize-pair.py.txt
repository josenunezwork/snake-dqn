def load(path):
    return json.loads(path.read_text())


def bind(path):
    assert path.is_file() and not path.is_symlink(), path
    with path.open("rb") as handle:
        digest = hashlib.file_digest(handle, "sha256").hexdigest()
    return {"path": str(path), "sha256": digest, "size": path.stat().st_size}


def validate():
    assert os.getpriority(os.PRIO_PROCESS, 0) == 15
    v = load(ROOT / "VERIFICATION-03.json")
    assert v["return_code"] == 0 and v["effective_niceness"] == 15
    assert all(value == "1" for value in v["numeric_threads"].values())
    assert v["pytorch_threads"] == v["pytorch_interop_threads"] == 1
    assert all(row["return_code"] == 0 for row in v["checks"])
    assert bind(Path(v["log"]["path"])) == v["log"]
    assert load(ROOT / "INDEPENDENT_REVIEW.json")["status"] == "NO_BLOCKING_FINDINGS"
    p = load(ROOT / "PREPARATION.json")
    m = load(ROOT / "manifest.json")
    l = load(LONG / "timing-operation.seal-draft.json")
    assert m["commit"] == l["commit"] == p["source_commit"]
    assert m["source_paths"] == l["source_paths"]
    assert m["f5_timing_synthetic_inventory"] == l["f5_timing_synthetic_inventory"]
    assert m["f5_timing_transfer_exception"] | {"context": "timing"} == l["f5_timing_transfer_exception"]
    assert bind(ROOT / "source.tar")["sha256"] == bind(LONG / "package/source.tar")["sha256"]
    for root, manifest in ((ROOT, m), (LONG, l)):
        for reference in load(root / "bindings.json").values():
            assert bind(Path(reference["path"])) == reference
        control = load(Path(next(r["path"] for r in manifest["inputs"] if r["dest"] == "control.json")))
        assert control["source_hashes"] == f5_package.source_inventory(BROKER)
        f5_job._source_admission(control)
        ledger = Path(next(r["path"] for r in manifest["inputs"] if r["dest"] == "verified-ledger.json"))
        assert not f5_timing_resume.load_ledger(ledger, p["stable_binding"])
        for row in manifest["inputs"]:
            if "path" in row:
                r = bind(Path(row["path"]))
                assert (r["sha256"], r["size"]) == (row["sha256"], row["size_bytes"])
        dispatch.require_resume_policy(manifest)
        dispatch.require_scientific_or_f5_plan_admission(manifest)
    supervisor.validate(m, BROKER)
    try:
        supervisor.validate(l, BROKER)
    except Exception as exc:
        assert "canary receipt is missing" in str(exc)
    else:
        raise AssertionError("long draft dispatchable")
    assert all(not Path(path).exists() for path in p["fresh_paths"])
    assert qualification.check(BROKER, Path(load(ROOT / "bindings.json")["qualified_registry"]["path"])) == p["cli_tree"]
    assert not subprocess.check_output(["git", "status", "--porcelain", "--untracked-files=all"], cwd=BROKER)
    assert not subprocess.check_output(["git", "branch", "--show-current"], cwd=BROKER).strip()
    assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=BROKER, text=True).strip() == m["commit"]
    return p

def publish():
    validate()
    assert not subprocess.check_output(["git", "status", "--porcelain", "--untracked-files=all"], cwd=REPO)
    for root, name in ((ROOT, "F5_ENVELOPE_CANARY04_20261026"), (LONG, "F5_ENVELOPE_LONGRUN01_DRAFT_20261026")):
        target = REPO / "research/bl_residual/ledger" / name
        assert not target.exists()
        seal = load(root / "SHA256SUMS.json")
        verify_export(root / "SHA256SUMS.json")
        members = dict(seal["files"])
        for name in seal["excluded_self_components"]:
            members[name] = {k:v for k,v in bind(root / name).items() if k != "path"}
        assert {p.relative_to(root).as_posix() for p in root.rglob("*") if p.is_file()} == set(members)
        records = {}
        for name, row in sorted(members.items()):
            source = root / name
            assert {k:v for k,v in bind(source).items() if k != "path"} == row
            record = {"external_path": str(source), **row}
            if row["size"] > 2_000_000:
                record["publication"] = "EXTERNAL_HASH_BOUND_ARTIFACT"
            else:
                relative = "sealed/" + name + (".txt" if name.endswith(".py") else "")
                destination = target / relative
                destination.parent.mkdir(parents=True, exist_ok=True)
                with destination.open("xb") as handle:
                    handle.write(source.read_bytes())
                assert bind(destination)["sha256"] == row["sha256"]
                record["published_relative_path"] = relative
            records[name] = record
        export_json(target / "ARTIFACT_PACKET.json", {"schema": "bl-f5-initial-pair-publication/v1", "package_root": str(root), "exhaustive_seal": bind(root / "SHA256SUMS.json"), "runtime_source_commit": load(ROOT / "PREPARATION.json")["source_commit"], "files": records, "script_snapshots": "Byte-identical .py.txt archive; no runtime source changes", "allocations": 0, "dispatches": 0, "provider_calls": 0})
        print(json.dumps({"publication": bind(target / "ARTIFACT_PACKET.json"), "files": len(records)}, indent=2))


if __name__ == "__main__":

    else:
